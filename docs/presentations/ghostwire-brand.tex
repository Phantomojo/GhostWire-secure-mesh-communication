% ── Images / diagrams ──────────────────────────────────────────
% Rendered mermaid diagrams arrive as PNGs. Constrain them to the text block
% so wide architecture diagrams scale down instead of running off the page.
% pandoc's own template defines \maxwidth; ours must too, and it must be a
% length assignment, not a \renewcommand.
\newcommand{\maxwidth}{0.95\linewidth}
\usepackage{graphicx}
\let\Oldincludegraphics\includegraphics
\renewcommand{\includegraphics}[2][]{%
  \IfNoValueTF{#1}%
    {\Oldincludegraphics[width=\maxwidth]{#2}}%
    {\Oldincludegraphics[#1,width=\maxwidth]{#2}}%
}
\usepackage{float}
% pandoc wraps every image in \pandocbounded{...}; without a definition for it,
% LaTeX silently discards the image. Provide a no-op passthrough.
\providecommand{\pandocbounded}[1]{#1}
% pandoc already emits a well-formed figure environment with \centering and a
% caption, and images are width-constrained above. Wrapping \figure again here
% breaks that structure, so leave it alone.
\usepackage{caption}
\captionsetup{font=small,labelfont={bf,color=GWDeep},skip=6pt}
\renewcommand{\figurename}{Diagram}

% ── pandoc helper macros ───────────────────────────────────────

% nested content render wrong.
\providecommand{\tightlist}{%

\providecommand{\passthrough}[1]{#1}
% Longtables: pandoc uses them for wide tables and needs these three.
\providecommand{\LTpost}{}
\providecommand{\LTend}{}
\setlength{\LTcapwidth}{0.95\textwidth}

% ── Tables ─────────────────────────────────────────────────────

% ── Deferred setup ─────────────────────────────────────────────
% Called from the document body (see the prelude the build script generates),
% because array/longtable/booktabs/tocloft are loaded after header-includes.
\newcommand{\gwdocsetup}{%
  % \arraystretch is intentionally not set here: it is only a valid length once
  % the array package is loaded, and pandoc's template loads array AFTER this
  % header-include. Setting it here raises "Illegal unit of measure" and aborts
  % the build. The template's own default is correct.
  \renewcommand{\cftsecfont}{\color{GWDeep}}%
  \renewcommand{\cftsecpagefont}{\color{GWGreen}}%
  \setlength{\cftbeforesecskip}{6pt}%
}
